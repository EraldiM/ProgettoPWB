CREATE TABLE coaches (
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(30) NOT NULL,
    last_name VARCHAR(30) NOT NULL,
    sex BOOL NOT NULL
);

CREATE TABLE categories (
    category_name VARCHAR(100) NOT NULL PRIMARY KEY
);

CREATE TABLE coaches_categories(
    id_coach INT NOT NULL;
    category_name VARCHAR(100) NOT NULL;
    PRIMARY KEY (id_coach, category_name)
    FOREIGN KEY(id_coach) REFERENCES coaches (id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY(category_name) REFERENCES categories (category_name) ON DELETE CASCADE ON UPDATE CASCADE
)
